Integrasi Google Sheets + Apps Script

Hubungkan HR Mailroom ke Google Sheets

Tempelkan URL Web App dari Google Apps Script di bawah ini agar setiap kali kamu menyimpan nomor surat di web, datanya otomatis terkirim ke Google Sheets (dan bisa ditarik kembali ke Buku Register).

Belum Terhubung

URL disimpan secara lokal di browser kamu. Jika dikosongkan, aplikasi tetap berjalan normal memakai penyimpanan browser & Export CSV.

1

Buat Tab Log_Surat

Di file Google Sheets kamu, pastikan ada tab bernama Log_Surat. Baris pertama akan otomatis diberi judul kolom oleh script saat pertama kali berjalan.

2

Deploy Web App

Di Extensions → Apps Script, tempel kode di bawah, klik Deploy → New deployment → Web app, pilih Execute as: Me dan Who has access: Anyone.

3

Tempel URL & Uji Coba

Salin URL berakhiran /exec ke kolom di atas, klik Simpan Koneksi, lalu coba buat 1 nomor surat baru.

Kode Google Apps Script (Kirim & Tarik Data Log_Surat) 
const SHEET_NAME = "Log_Surat";

function getOrCreateSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
  }
  if (sheet.getLastRow() === 0) {
    sheet.appendRow([
      "ID",
      "Timestamp",
      "Nomor Surat",
      "Tanggal Surat",
      "Tahun",
      "Kode Awal",
      "Kelompok",
      "Kode Jenis",
      "Divisi",
      "Entitas",
      "Nama Penerima",
      "Keterangan / Perihal",
      "Notes",
      "Link Draft"
    ]);
  }
  return sheet;
}

function doPost(e) {
  try {
    const sheet = getOrCreateSheet_();
    const data = JSON.parse(e.postData.contents);
    const id = data.id || ("row-" + new Date().getTime());

    sheet.appendRow([
      id,
      new Date().toISOString(),
      data.fullNumber || "",
      data.date || "",
      data.year || "",
      data.prefix || "",
      data.category || "",
      data.code || "",
      data.division || "",
      data.entity || "",
      data.recipient || "",
      data.subject || "",
      data.notes || "",
      data.docUrl || ""
    ]);

    return ContentService
      .createTextOutput(JSON.stringify({ status: "success", id: id }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    const sheet = getOrCreateSheet_();
    const values = sheet.getDataRange().getValues();
    if (values.length <= 1) {
      return ContentService
        .createTextOutput(JSON.stringify({ status: "success", logs: [] }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const logs = [];
    for (let i = 1; i < values.length; i++) {
      const r = values[i];
      if (!r[2]) continue;
      const fullNum = String(r[2]);
      const seqMatch = fullNum.match(/^[AB]\.(\d+)/i);
      logs.push({
        id: String(r[0] || ("gs-" + i)),
        patternType: "ADMIN",
        prefix: String(r[5] || "A"),
        seq: seqMatch ? seqMatch[1] : "01",
        division: String(r[8] || "HRD"),
        holding: "",
        entity: String(r[9] || "EVO"),
        category: String(r[6] || "SK"),
        code: String(r[7] || "SK"),
        date: String(r[3] || "").slice(0, 10),
        year: Number(r[4]) || 2027,
        fullNumber: fullNum,
        recipient: String(r[10] || "-"),
        subject: String(r[11] || "-"),
        notes: String(r[12] || ""),
        docUrl: String(r[13] || "")
      });
    }

    return ContentService
      .createTextOutput(JSON.stringify({ status: "success", logs: logs.reverse() }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
